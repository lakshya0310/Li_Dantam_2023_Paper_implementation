cmake_minimum_required(VERSION 3.10)
project(infeasibility_pipeline)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Eigen3  REQUIRED)
find_package(OpenMP  REQUIRED)
find_package(GUDHI   REQUIRED)
find_package(ompl    REQUIRED)
find_package(CUDAToolkit REQUIRED)

# NLopt — not in every distro's cmake registry; fall back to pkg-config
find_package(NLopt QUIET)
if(NOT NLopt_FOUND)
    find_library(NLOPT_LIB nlopt HINTS /usr/local/lib /usr/lib)
    find_path   (NLOPT_INC nlopt.hpp HINTS /usr/local/include /usr/include)
    if(NOT NLOPT_LIB OR NOT NLOPT_INC)
        message(FATAL_ERROR "NLopt not found — install libnlopt-dev or set NLOPT_DIR")
    endif()
endif()

include_directories(include ../include)
link_directories(/usr/local/lib)

add_definitions(-DELPP_DISABLE_LOGS)

add_executable(infeasibility_pipeline
    infeasibility_pipeline.cpp
    src/man_trace.cpp
    src/prm_sampler.cpp
    src/svm.cpp
    src/out.c           # compiled robot scene graph (generated by aarxc)
)

target_include_directories(infeasibility_pipeline PRIVATE
    ${EIGEN3_INCLUDE_DIR}
    ${OMPL_INCLUDE_DIRS}
    /usr/include/gudhi
    /usr/local/include/amino-1.0   # amino headers
    ../libcuckoo                   # libcuckoo (header-only)
    ${CUDAToolkit_INCLUDE_DIRS}
    # NLopt header (if found via fallback)
    $<$<NOT:$<TARGET_EXISTS:NLopt::nlopt>>:${NLOPT_INC}>
)

target_link_libraries(infeasibility_pipeline PRIVATE
    OpenMP::OpenMP_CXX
    Eigen3::Eigen
    ${OMPL_LIBRARIES}
    amino
    amino-planning
    amino-collision
    thundersvm          # ThunderSVM — installed to /usr/local/lib
    # NLopt
    $<IF:$<TARGET_EXISTS:NLopt::nlopt>,NLopt::nlopt,${NLOPT_LIB}>
    CUDA::cudart
)
